import { CodePreview, CodePreviewGroup, ChipComponent } from '@site/src/components';

# Usage guidelines

## Labels

Truncate labels that are longer than 20 characters and do not wrap text on multiple lines.

<CodePreviewGroup>
  <CodePreview
    scriptPaths={[
      'https://unpkg.com/@popperjs/core@2',
      '/scripts/components/feedback-tooltip.js'
    ]}
    hideCode
    minHeight="192px"
    guideline='permitted'
    code={ChipComponent({
      content: 'Ettevõtlus, kaubandus',
      variant: 'selection',
    })}
  />
  <CodePreview
    hideCode
    minHeight="192px"
    guideline='prohibited'
    code={[
      {
        code:
          `<div style="display: flex; flex-direction: column; align-items: center; justify-content: center; gap: 16px;">` +
          ChipComponent({
            content: 'Ettevõtlus, kaubandus ja reklaam',
            truncate: false,
            customTextStyle: 'min-width: 260px;',
            variant: 'selection',
          }) +
          ChipComponent({
            content: 'Ettevõtlus, kaubandus ja reklaam',
            truncate: false,
            customTextStyle: 'display: flex; max-width: 176px; height: 48px; align-items: center;',
            variant: 'selection',
          }) +
          `</div>`
      },
  ]}/>
</CodePreviewGroup>

## Size

Do not use different sizes in a group.
